class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        std::unordered_map<int, int> streakMap;
        int longestStreak = 0;

        for(int num : nums) {

            if(streakMap.find(num) != streakMap.end()) {
                continue;
            }

            int left = streakMap.count(num - 1) ? streakMap[num -1] : 0;
            int right = streakMap.count(num + 1) ? streakMap[num + 1] : 0;

            int currentStreak = left + right +1;
            longestStreak = std::max(longestStreak, currentStreak);

            streakMap[num] = currentStreak;

            streakMap[num - left] = currentStreak;
            streakMap[num + right] = currentStreak;
        }

    return longestStreak;
        
    }
};
