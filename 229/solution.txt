class Solution {
public:
    vector<int> majorityElement(vector<int>& nums) {
        int n = nums.size();
        int threshold = n/3;
        unordered_map<int, int>freq;
        vector<int> res;

        for(int num : nums) {
            freq[num]++;

        }

        for(const auto& [num, count] : freq){
            if (count > threshold) {
                res.push_back(num);
            }

        }

        return res;
        
    }
};
